```html
<div class="dc-overlay cart-modal-overlay"
     id="cartModalOverlay"
     data-cart-modal-overlay
     hidden
     aria-hidden="true"
     role="dialog"
     aria-modal="true"
     aria-labelledby="cartModalTitle">

  <div class="dc-modal cart-modal">
    <div class="dc-body cart-modal__body">

      <h3 class="dc-title cart-modal__title" id="cartModalTitle">
        {lang === 'es' ? 'Tu selección de servicios' : 'Your service selection'}
      </h3>

      <!-- Item list: populated by cart.ts openCartModal() -->
      <div class="cart-modal__list"
           aria-label={lang === 'es' ? 'Selección' : 'Selection'}>
      </div>

      <!-- Total line: populated by cart.ts -->
      <div class="cart-modal__total">
      </div>

      <!-- Disclaimer -->
      <p class="cart-modal__disclaimer">
        {lang === 'es'
          ? '(Precios sujetos a cotización final)'
          : '(Prices subject to final quote)'}
      </p>

      <!-- Action buttons -->
      <div class="cart-modal__actions">
        <button class="cart-modal__copy" type="button" data-cart-copy>
          {lang === 'es' ? 'Copiar selección' : 'Copy selection'}
        </button>
        <a class="cart-modal__discord"
           href="https://discord.gg/DSMmwU35cs"
           target="_blank"
           rel="noopener noreferrer">
          {lang === 'es' ? 'Ir a Discord' : 'Go to Discord'}
        </a>
      </div>

      <!-- Close button — reuses dc-dismiss styles from chrome.css -->
      <button class="dc-dismiss" type="button" data-cart-modal-dismiss>
        {lang === 'es' ? 'Cerrar' : 'Close'}
      </button>

    </div>
  </div>

</div>
```
